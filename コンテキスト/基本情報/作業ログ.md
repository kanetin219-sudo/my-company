### Code.gs（完成済み）
```javascript
const FOLDER_ID = "ここにGoogleドライブのフォルダIDを入れる"; // 要変更

function doGet() {
  return HtmlService.createHtmlOutputFromFile('index')
    .setTitle('ANELLA CAFE 入場フォーム')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

function saveForm(formData) {
  try {
    const folder = DriveApp.getFolderById(FOLDER_ID);
    const doc = DocumentApp.create('ANELLA_temp');
    const body = doc.getBody();
    body.getParagraphs()[0].removeFromParent();

    const title = body.appendParagraph('ANELLA CAFE　ご利用時の注意事項');
    title.setHeading(DocumentApp.ParagraphHeading.HEADING1);
    title.setAlignment(DocumentApp.HorizontalAlignment.CENTER);
    body.appendParagraph('');

    const info = [
      ['来店日', formData.date],
      ['来店人数', `大人 ${formData.adult}人　小学生以上 ${formData.child}人　未就学 ${formData.infant}人`],
      ['ご利用回数', formData.visit],
      ['お気に入りの子', formData.favorite || '―'],
    ];

    const table = body.appendTable();
    info.forEach(([label, value]) => {
      const row = table.appendTableRow();
      row.appendTableCell(label).setWidth(120);
      row.appendTableCell(value);
    });

    body.appendParagraph('');
    body.appendParagraph('▼ ご利用時の注意事項に同意します。').setBold(true);
    body.appendParagraph('');

    formData.signatures.forEach((sig, i) => {
      if (sig && sig.startsWith('data:image')) {
        body.appendParagraph(`署名 ${i + 1}`).setBold(true);
        const base64 = sig.split(',')[1];
        const imgBlob = Utilities.newBlob(
          Utilities.base64Decode(base64), 'image/png', `sig${i}.png`
        );
        body.appendImage(imgBlob).setWidth(300).setHeight(120);
        body.appendParagraph('');
      }
    });

    doc.saveAndClose();
    const dateStr = formData.date.replace(/\//g, '-');
    const pdfBlob = DriveApp.getFileById(doc.getId())
      .getAs('application/pdf')
      .setName(`${dateStr}_入場記録.pdf`);
    folder.createFile(pdfBlob);
    DriveApp.getFileById(doc.getId()).setTrashed(true);
    return JSON.stringify({ success: true });
  } catch (e) {
    return JSON.stringify({ success: false, error: e.message });
  }
}
```
